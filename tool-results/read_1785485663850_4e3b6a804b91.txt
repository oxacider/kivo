    60→    60→function MessageReactions({
    61→    61→  reactions,
    62→    62→  myUserId,
    63→    63→  onReact,
    64→    64→  onLongPress,
    65→    65→}: {
    66→    66→  reactions: Reaction[];

    68→    68→  onReact: (emoji: string) => void;
    69→    69→  onLongPress?: () => void;
    70→    70→}) {
    71→    71→  if (!reactions || reactions.length === 0) return null;
    72→    72→
    73→    73→  const grouped = reactions.reduce<Record<string, Reaction[]>>((acc, r) => {
    74→    74→    if (!acc[r.emoji]) acc[r.emoji] = [];
    75→    75→    acc[r.emoji].push(r);
    76→    76→    return acc;
    77→    77→  }, {});

    79→    79→  return (
    80→    80→    <div className="flex flex-wrap gap-1 mt-0.5">
    81→    81→      {Object.entries(grouped).map(([emoji, rs]) => {
    82→    82→        const isMine = rs.some(r => r.userId === myUserId);
    83→    83→        const count = rs.length;
    84→    84→        const names = rs.map(r => r.user?.displayName).filter(Boolean).join(', ');
    85→    85→        return (
    86→    86→          <motion.button
    87→    87→            key={emoji}
    88→    88→            initial={{ scale: 0, opacity: 0 }}
    89→    89→            animate={{ scale: 1, opacity: 1 }}
    90→    90→            transition={{ type: 'spring', stiffness: 500, damping: 25 }}
    91→    91→            whileTap={{ scale: 0.85 }}
    92→    92→            onClick={(e) => { e.stopPropagation(); onReact(emoji); }}
    93→    93→            title={names}
    94→    94→            className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-xs transition-colors ${
    95→    95→              isMine
    96→    96→                ? 'bg-primary/15 ring-1 ring-primary/30'
    97→    97→                : 'bg-surface-2 hover:bg-surface-hover'
    98→    98→            }`}
    99→    99→          >
   100→   100→            <span className="text-sm leading-none">{emoji}</span>
   101→   101→            {count > 1 && <span className="text-[10px] font-medium text-muted-foreground">{count}</span>}
   102→   102→          </motion.button>
   103→   103→        );

   190→   190→  const inputRef = useRef<HTMLTextAreaElement>(null);
   191→   191→  const typingTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
   192→   192→  const [wasAtBottom, setWasAtBottom] = useState(true);
   193→   193→
   194→   194→  const activeConv = conversations.find((c) => c.id === activeConversationId);
   195→   195→  const otherUser = activeConv?.otherUser;
   196→   196→  const activeTypingUsers = useMemo(
   197→   197→    () => typingUsers.filter(t => t.conversationId === activeConversationId && t.isTyping && t.userId !== user?.id),
   198→   198→    [typingUsers, activeConversationId, user?.id]

   201→   201→  const typingDisplay = activeTypingUsers.length === 1
   202→   202→    ? activeTypingUsers[0]

   204→   204→
   205→   205→  // Track scroll position

   207→   207→    const el = scrollRef.current;
   208→   208→    if (!el) return;
   209→   209→    const atBottom = el.scrollHeight - el.scrollTop - el.clientHeight < 80;
   210→   210→    setWasAtBottom(atBottom);
   211→   211→  }, []);
   212→   212→
   213→   213→  // Socket listeners for delivered, read, reactions, and optimistic message replacement
   214→   214→  useEffect(() => {
   215→   215→    if (!token || token.startsWith('demo-')) return;
   216→   216→    const socket = getSocket();
   217→   217→
   218→   218→    const onDelivered = (data: { messageId: string; conversationId: string; status: string }) => {
   219→   219→      if (data.conversationId !== activeConversationId) return;
   220→   220→      // Find the optimistic sending message and replace with delivered
   221→   221→      const msgs = useChatStore.getState().messages;
   222→   222→      const sending = msgs.find(m => m.status === 'sending' && m.conversationId === data.conversationId);
   223→   223→      if (sending) {
   224→   224→        // Will be updated when the real message:new comes from server
   225→   225→      }
   226→   226→      updateMessage(data.messageId, { status: 'delivered' });
   227→   227→    };
   228→   228→
   229→   229→    const onRead = (data: { conversationId: string; userId: string }) => {
   230→   230→      if (data.conversationId !== activeConversationId) return;
   231→   231→      // Mark all sent/delivered messages as read
   232→   232→      const msgs = useChatStore.getState().messages;
   233→   233→      msgs.forEach(m => {
   234→   234→        if (m.senderId === user?.id && (m.status === 'sent' || m.status === 'delivered' || m.status === 'sending')) {
   235→   235→          updateMessage(m.id, { status: 'read' });
   236→   236→        }
   237→   237→      });
   238→   238→    };
   239→   239→
   240→   240→    const onNew = (msg: any) => {
   241→   241→      // Replace optimistic sending message with the real server message
   242→   242→      const msgs = useChatStore.getState().messages;
   243→   243→      const sending = msgs.find(m => m.status === 'sending' && m.conversationId === msg.conversationId && m.content === msg.content && m.senderId === msg.senderId);
   244→   244→      if (sending && sending.id.startsWith('temp-')) {
   245→   245→        // Remove the temp message, the real one will be added by the store's addMessage
   246→   246→        useChatStore.getState().removeMessage(sending.id);
   247→   247→      }
   248→   248→    };
   249→   249→
   250→   250→    const onReaction = (data: any) => {
   251→   251→      if (data.removed) {
   252→   252→        removeReaction(data.messageId, data.userId, data.emoji);
   253→   253→      } else {
   254→   254→        addReaction(data.messageId, data);
   255→   255→      }
   256→   256→    };
   257→   257→
   258→   258→    socket.on('message:delivered', onDelivered);
   259→   259→    socket.on('message:read', onRead);
   260→   260→    socket.on('message:new', onNew);
   261→   261→    socket.on('reaction:update', onReaction);
   262→   262→
   263→   263→    return () => {
   264→   264→      socket.off('message:delivered', onDelivered);
   265→   265→      socket.off('message:read', onRead);
   266→   266→      socket.off('message:new', onNew);
   267→   267→      socket.off('reaction:update', onReaction);
   268→   268→    };
   269→   269→  }, [token, activeConversationId, user?.id, updateMessage, removeMessage, addReaction, removeReaction]);
   270→   270→
   271→   271→  // Load messages when conversation changes
   272→   272→  useEffect(() => {
   273→   273→    if (!activeConversationId || !token) return;
   274→   274→    const isDemo = token.startsWith('demo-');
   275→   275→    if (isDemo) return;

   278→   278→      try {
   279→   279→        const msgs = await api<Message[]>('/conversations/' + activeConversationId + '/messages', { token });

   281→   281→      } catch (err: any) { if (!cancelled) toast.error(err.message); }

   283→   283→    return () => { cancelled = true; };

   286→   286→  // Scroll to bottom on new messages (only if already at bottom)

   290→   290→        scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: 'smooth' });

   314→   314→      updatedAt: new Date().toISOString(),
   315→   315→      sender: user ? { id: user.id, email: user.email, displayName: user.displayName, username: user.username, avatar: user.avatar, bio: user.bio, status: user.status, online: user.online, lastSeen: user.lastSeen, theme: user.theme, emailVerified: user.emailVerified, showOnline: user.showOnline, showLastSeen: user.showLastSeen, showReadReceipts: user.showReadReceipts, createdAt: user.createdAt, updatedAt: user.updatedAt } : undefined,
