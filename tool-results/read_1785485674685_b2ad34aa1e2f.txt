   237→      });
   238→    };

   240→    const onNew = (msg: any) => {

   264→      socket.off('message:delivered', onDelivered);
   265→      socket.off('message:read', onRead);
   266→      socket.off('message:new', onNew);

   643→                        <button onClick={(e) => { e.stopPropagation(); setReplyTo(msg); inputRef.current?.focus(); setQuickReactMsgId(null); }} className="rounded-md bg-popover p-1 shadow-sm border border-border/50 hover:bg-surface-hover transition-colors">
   644→                          <CornerDownRight className="h-3 w-3 text-muted-foreground" />
   645→                        </button>
   646→                        <button onClick={(e) => { e.stopPropagation(); forwardMessage(msg); }} className="rounded-md bg-popover p-1 shadow-sm border border-border/50 hover:bg-surface-hover transition-colors">
   647→                          <Forward className="h-3 w-3 text-muted-foreground" />
   648→                        </button>
   649→                        <button onClick={(e) => { e.stopPropagation(); startEdit(msg); }} className="rounded-md bg-popover p-1 shadow-sm border border-border/50 hover:bg-surface-hover transition-colors">
   650→                          <Edit3 className="h-3 w-3 text-muted-foreground" />
   651→                        </button>
   652→                        <button onClick={(e) => { e.stopPropagation(); deleteMessage(msg.id); }} className="rounded-md bg-popover p-1 shadow-sm border border-border/50 hover:bg-destructive/10 transition-colors">
   653→                          <Trash2 className="h-3 w-3 text-muted-foreground" />
   654→                        </button>
   655→                      </div>
   656→                    )}
   657→                    {!isMine && !msg.deleted && isLast && (
   658→                      <div className="absolute -bottom-3 left-0 hidden group-hover:flex items-center gap-0.5">
   659→                        <button onClick={(e) => { e.stopPropagation(); copyMessage(msg.content); }} className="rounded-md bg-popover p-1 shadow-sm border border-border/50 hover:bg-surface-hover transition-colors">
   660→                          <Copy className="h-3 w-3 text-muted-foreground" />
   661→                        </button>
   662→                        <button onClick={(e) => { e.stopPropagation(); setReplyTo(msg); inputRef.current?.focus(); setQuickReactMsgId(null); }} className="rounded-md bg-popover p-1 shadow-sm border border-border/50 hover:bg-surface-hover transition-colors">
   663→                          <CornerDownRight className="h-3 w-3 text-muted-foreground" />
   664→                        </button>
   665→                        <button onClick={(e) => { e.stopPropagation(); forwardMessage(msg); }} className="rounded-md bg-popover p-1 shadow-sm border border-border/50 hover:bg-surface-hover transition-colors">
   666→                          <Forward className="h-3 w-3 text-muted-foreground" />
   667→                        </button>
   668→                      </div>
   669→                    )}
   670→                  </div>
   671→
   672→                  {/* Failed message retry */}
   673→                  {isFailed && isLast && (
   674→                    <div className="mt-1 flex justify-end">
   675→                      <button
   676→                        onClick={() => retryMessage(msg)}
   677→                        className="flex items-center gap-1 rounded-lg bg-destructive/10 px-2 py-1 text-[10px] font-medium text-destructive hover:bg-destructive/20 transition-colors"
   678→                      >
   679→                        <RotateCcw className="h-3 w-3" /> Retry
   680→                      </button>
   681→                    </div>
   682→                  )}
   683→
   684→                  {/* Reactions display */}
   685→                  {msg.reactions && msg.reactions.length > 0 && (
   686→                    <div className={isMine ? 'flex justify-end' : ''}>
   687→                      <MessageReactions
   688→                        reactions={msg.reactions}
   689→                        myUserId={user?.id || ''}
   690→                        onReact={(emoji) => handleReact(msg.id, emoji)}
   691→                      />
   692→                    </div>
   693→                  )}
   694→
   695→                  {/* Status & time */}
   696→                  {isLast && (
   697→                    <div className={`mt-0.5 flex items-center gap-1 ${isMine ? 'justify-end' : 'justify-start'} px-1`}>
   698→                      <span className="text-[10px] text-muted-foreground/50">
   699→                        {format(new Date(msg.createdAt), 'h:mm a')}
   700→                      </span>
   701→                      {isMine && <MessageStatus status={msg.status} />}
   702→                    </div>
   703→                  )}
   704→                </div>
   705→              </motion.div>
   706→            );
   707→          })}
   708→
   709→          {/* ========== PREMIUM TYPING INDICATOR ========== */}
   710→          <AnimatePresence>
   711→            {isTyping && (
   712→              <motion.div
   713→                initial={{ opacity: 0, y: 8 }}
   714→                animate={{ opacity: 1, y: 0 }}
   715→                exit={{ opacity: 0, y: 8 }}
   716→                transition={{ duration: 0.2 }}
   717→                className="mt-2 flex items-center gap-2.5"
   718→              >
   719→                {typingDisplay?.user?.avatar && (
   720→                  <Avatar className="h-6 w-6">
   721→                    <AvatarImage src={typingDisplay.user.avatar || undefined} />
   722→                    <AvatarFallback className="text-[8px] bg-primary/10 text-primary">{getInitials(typingDisplay.user.displayName || '?')}</AvatarFallback>
   723→                  </Avatar>
   724→                )}
   725→                <div className="flex items-center gap-2">
   726→                  <div className="flex items-center gap-1 rounded-2xl rounded-bl-md bg-surface-2 px-4 py-2.5">
   727→                    {[0, 1, 2].map((i) => (
   728→                      <span
   729→                        key={i}
   730→                        className="h-[6px] w-[6px] rounded-full bg-muted-foreground/50"
   731→                        style={{
   732→                          animation: 'kivo-typing-dot 1.4s ease-in-out infinite',
   733→                          animationDelay: `${i * 0.2}s`,
   734→                        }}
   735→                      />
   736→                    ))}
   737→                  </div>
   738→                  <span className="text-[11px] text-muted-foreground/60 font-medium">
   739→                    {typingDisplay?.user?.displayName || 'Someone'} is typing
   740→                  </span>
   741→                </div>
   742→              </motion.div>
   743→            )}
   744→          </AnimatePresence>
   745→        </div>
   746→      </div>
   747→
   748→      {/* ========== REPLY PREVIEW BAR ========== */}
   749→      <AnimatePresence>
   750→        {replyTo && (
   751→          <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: 'auto', opacity: 1 }} exit={{ height: 0, opacity: 0 }} className="overflow-hidden border-t border-border/30 bg-surface-1">
   752→            <div className="flex items-center gap-2 px-4 py-2">
   753→              <div className="w-1 h-8 rounded-full bg-primary" />
   754→              <div className="flex-1 min-w-0">
   755→                <p className="text-[10px] font-medium text-primary">Reply to {replyTo.senderId === user?.id ? 'yourself' : otherUser?.displayName}</p>
   756→                <p className="text-[11px] text-muted-foreground truncate">{replyTo.deleted ? 'Message deleted' : replyTo.content}</p>
   757→              </div>
   758→              <button onClick={() => setReplyTo(null)} className="text-muted-foreground hover:text-foreground"><X className="h-4 w-4" /></button>
   759→            </div>
   760→          </motion.div>
   761→        )}
   762→      </AnimatePresence>
   763→
   764→      {/* ========== FORWARD PREVIEW BAR ========== */}
   765→      <AnimatePresence>
   766→        {forwardMsg && (
   767→          <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: 'auto', opacity: 1 }} exit={{ height: 0, opacity: 0 }} className="overflow-hidden border-t border-border/30 bg-surface-1">
   768→            <div className="flex items-center gap-2 px-4 py-2">
   769→              <Forward className="h-3.5 w-3.5 text-primary" />
   770→              <div className="flex-1 min-w-0">
   771→                <p className="text-[10px] font-medium text-primary">Forward message</p>
   772→                <p className="text-[11px] text-muted-foreground truncate">{forwardMsg.content}</p>
   773→              </div>
   774→              <button onClick={() => setForwardMsg(null)} className="text-muted-foreground hover:text-foreground"><X className="h-4 w-4" /></button>
   775→            </div>
   776→          </motion.div>
   777→        )}
   778→      </AnimatePresence>
   779→
   780→      {/* ========== EDITING INDICATOR ========== */}
   781→      <AnimatePresence>
   782→        {editingId && (
   783→          <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: 'auto', opacity: 1 }} exit={{ height: 0, opacity: 0 }} className="overflow-hidden border-t border-border/30 bg-surface-1">
   784→            <div className="flex items-center gap-2 px-4 py-2">
   785→              <Edit3 className="h-3.5 w-3.5 text-primary" />
   786→              <span className="text-xs text-primary font-medium">Editing message</span>
   787→              <div className="flex-1" />
   788→              <button onClick={() => { setEditingId(null); setEditText(''); }} className="text-muted-foreground hover:text-foreground"><X className="h-4 w-4" /></button>
   789→            </div>
   790→          </motion.div>
   791→        )}
   792→      </AnimatePresence>
   793→
   794→      {/* ========== EMOJI PICKER ========== */}
   795→      <AnimatePresence>
   796→        {showEmoji && (
   797→          <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: 'auto', opacity: 1 }} exit={{ height: 0, opacity: 0 }} className="overflow-hidden border-t border-border/30 bg-surface-1">
   798→            <div className="flex flex-wrap gap-1.5 px-4 py-3 max-h-36 overflow-y-auto">
   799→              {EMOJI_LIST.map((emoji) => (
   800→                <button
   801→                  key={emoji}
   802→                  onClick={() => { setInput((p) => p + emoji); inputRef.current?.focus(); }}
   803→                  className="flex h-8 w-8 items-center justify-center rounded-lg hover:bg-surface-hover text-lg transition-colors"
   804→                >
   805→                  {emoji}
   806→                </button>
   807→              ))}
   808→            </div>
   809→          </motion.div>
   810→        )}
   811→      </AnimatePresence>
   812→
   813→      {/* ========== INPUT BAR ========== */}
   814→      <div className="border-t border-border/30 bg-surface-1 px-4 py-3">
   815→        <div className="mx-auto flex max-w-2xl items-end gap-2">
   816→          {/* Media attach buttons (architecture prep) */}
   817→          <div className="hidden sm:flex items-center gap-1">
   818→            <button className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl text-muted-foreground hover:text-foreground hover:bg-surface-hover transition-colors" title="Attach image">
   819→              <ImageIcon className="h-4.5 w-4.5" />
   820→            </button>
   821→            <button className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl text-muted-foreground hover:text-foreground hover:bg-surface-hover transition-colors" title="Attach file">
   822→              <Paperclip className="h-4.5 w-4.5" />
   823→            </button>
   824→            <button className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl text-muted-foreground hover:text-foreground hover:bg-surface-hover transition-colors" title="Voice message">
   825→              <Mic className="h-4.5 w-4.5" />
   826→            </button>
   827→          </div>
   828→
   829→          <button
   830→            onClick={() => setShowEmoji(!showEmoji)}
   831→            className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-xl transition-colors ${
   832→              showEmoji ? 'bg-primary/10 text-primary' : 'text-muted-foreground hover:text-foreground hover:bg-surface-hover'
   833→            }`}
   834→          >
   835→            <Smile className="h-4.5 w-4.5" />
   836→          </button>
   837→          <div className="flex-1 relative">
   838→            <textarea
   839→              ref={inputRef}
   840→              value={editingId ? editText : input}
   841→              onChange={editingId ? (e) => setEditText(e.target.value) : handleInputChange}
   842→              onKeyDown={handleKeyDown}
   843→              placeholder="Type a message..."
   844→              rows={1}
   845→              className="w-full resize-none rounded-xl bg-surface-2 border border-border/30 px-4 py-2.5 text-sm leading-relaxed placeholder:text-muted-foreground/40 focus:outline-none focus:ring-1 focus:ring-primary/30 max-h-[120px]"
   846→            />
   847→          </div>
   848→          <button
   849→            onClick={editingId ? saveEdit : forwardMsg ? sendForward : sendMessage}
   850→            disabled={editingId ? !editText.trim() : forwardMsg ? false : !input.trim()}
   851→            className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl bg-primary text-primary-foreground transition-all hover:opacity-90 disabled:opacity-40 kivo-glow"
   852→          >
   853→            {editingId ? <Check className="h-4 w-4" /> : forwardMsg ? <Forward className="h-4 w-4" /> : <Send className="h-4 w-4" />}
   854→          </button>
   855→        </div>
   856→      </div>
   857→    </div>
   858→  );
   859→}
