    37→// --- Message Status Icon ---
    38→function MessageStatus({ status, className = '' }: { status: Message['status']; className?: string }) {
    39→  if (status === 'failed') {
    40→    return <AlertCircle className={`h-3.5 w-3.5 text-destructive ${className}`} />;
    41→  }
    42→  if (status === 'sending') {
    43→    return <div className={`flex items-center gap-0.5 ${className}`}>
    44→      {[0, 1, 2].map(i => (
    45→        <span key={i} className="h-1 w-1 rounded-full bg-current opacity-40"
    46→          style={{ animation: 'kivo-status-pulse 1s ease-in-out infinite', animationDelay: `${i * 0.2}s` }} />
    47→      ))}
    48→    </div>;
    49→  }
    50→  if (status === 'read') {
    51→    return <CheckCheck className={`h-3.5 w-3.5 text-primary ${className}`} />;
    52→  }
    53→  if (status === 'delivered') {
    54→    return <CheckCheck className={`h-3.5 w-3.5 text-muted-foreground/50 ${className}`} />;
    55→  }
    56→  return <Check className={`h-3 w-3 text-muted-foreground/40 ${className}`} />;
    57→}

    59→// --- Reaction Badge on message ---
    60→function MessageReactions({
    61→  reactions,
    62→  myUserId,
    63→  onReact,
    64→  onLongPress,
    65→}: {
    66→  reactions: Reaction[];
    67→  myUserId: string;
    68→  onReact: (emoji: string) => void;
    69→  onLongPress?: () => void;
    70→}) {
    71→  if (!reactions || reactions.length === 0) return null;
    72→
    73→  const grouped = reactions.reduce<Record<string, Reaction[]>>((acc, r) => {
    74→    if (!acc[r.emoji]) acc[r.emoji] = [];
    75→    acc[r.emoji].push(r);
    76→    return acc;
    77→  }, {});
    78→
    79→  return (
    80→    <div className="flex flex-wrap gap-1 mt-0.5">
    81→      {Object.entries(grouped).map(([emoji, rs]) => {
    82→        const isMine = rs.some(r => r.userId === myUserId);
    83→        const count = rs.length;
    84→        const names = rs.map(r => r.user?.displayName).filter(Boolean).join(', ');
    85→        return (
    86→          <motion.button
    87→            key={emoji}
    88→            initial={{ scale: 0, opacity: 0 }}
    89→            animate={{ scale: 1, opacity: 1 }}
    90→            transition={{ type: 'spring', stiffness: 500, damping: 25 }}
    91→            whileTap={{ scale: 0.85 }}
    92→            onClick={(e) => { e.stopPropagation(); onReact(emoji); }}
    93→            title={names}
    94→            className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-xs transition-colors ${
    95→              isMine
    96→                ? 'bg-primary/15 ring-1 ring-primary/30'
    97→                : 'bg-surface-2 hover:bg-surface-hover'
    98→            }`}
    99→          >
   100→            <span className="text-sm leading-none">{emoji}</span>
   101→            {count > 1 && <span className="text-[10px] font-medium text-muted-foreground">{count}</span>}
   102→          </motion.button>
   103→        );
   104→      })}
   105→    </div>
   106→  );
   107→}

   177→export function ConversationView() {
   178→  const { activeConversationId, conversations, messages, setMessages, addMessage, updateMessage, removeMessage, typingUsers, setActiveConversationId, updateConversation, addReaction, removeReaction } = useChatStore();
   179→  const { user, token } = useAuthStore();
   180→  const [input, setInput] = useState('');
   181→  const [showEmoji, setShowEmoji] = useState(false);
   182→  const [replyTo, setReplyTo] = useState<Message | null>(null);
   183→  const [editingId, setEditingId] = useState<string | null>(null);
   184→  const [editText, setEditText] = useState('');
   185→  const [forwardMsg, setForwardMsg] = useState<Message | null>(null);
   186→  const [showSearch, setShowSearch] = useState(false);
   187→  const [searchMsg, setSearchMsg] = useState('');
   188→  const [quickReactMsgId, setQuickReactMsgId] = useState<string | null>(null);
   189→  const scrollRef = useRef<HTMLDivElement>(null);
   190→  const inputRef = useRef<HTMLTextAreaElement>(null);
   191→  const typingTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
   192→  const [wasAtBottom, setWasAtBottom] = useState(true);
   193→
   194→  const activeConv = conversations.find((c) => c.id === activeConversationId);
   195→  const otherUser = activeConv?.otherUser;
   196→  const activeTypingUsers = useMemo(
   197→    () => typingUsers.filter(t => t.conversationId === activeConversationId && t.isTyping && t.userId !== user?.id),
   198→    [typingUsers, activeConversationId, user?.id]
   199→  );
   200→  const isTyping = activeTypingUsers.length > 0;
   201→  const typingDisplay = activeTypingUsers.length === 1
   202→    ? activeTypingUsers[0]
   203→    : null;
   204→
   205→  // Track scroll position
   206→  const handleScroll = useCallback(() => {
   207→    const el = scrollRef.current;
   208→    if (!el) return;
   209→    const atBottom = el.scrollHeight - el.scrollTop - el.clientHeight < 80;
   210→    setWasAtBottom(atBottom);
   211→  }, []);
   212→
   213→  // Socket listeners for delivered, read, reactions, and optimistic message replacement
   214→  useEffect(() => {
   215→    if (!token || token.startsWith('demo-')) return;
   216→    const socket = getSocket();
   217→
   218→    const onDelivered = (data: { messageId: string; conversationId: string; status: string }) => {
   219→      if (data.conversationId !== activeConversationId) return;
   220→      updateMessage(data.messageId, { status: 'delivered' });
   221→    };
   222→
   223→    const onRead = (data: { conversationId: string; userId: string }) => {
   224→      if (data.conversationId !== activeConversationId) return;
   225→      // Batch: mark all sent/delivered/sending messages as read in one update
   226→      const msgs = useChatStore.getState().messages;
   227→      const updates = msgs
   228→        .filter(m => m.senderId === user?.id && (m.status === 'sent' || m.status === 'delivered' || m.status === 'sending'))
   229→        .map(m => ({ id: m.id, status: 'read' as const }));
   230→      if (updates.length > 0) {
   231→        useChatStore.setState((state) => ({
   232→          messages: state.messages.map((m) => {
   233→            const u = updates.find((up) => up.id === m.id);
   234→            return u ? { ...m, status: u.status } : m;
   235→          }),
   236→        }));
   237→      }
   238→    };
   239→
   240→    const onNew = (msg: any) => {
   241→      // Replace optimistic sending message with the real server message
   242→      const msgs = useChatStore.getState().messages;
   243→      const sending = msgs.find(m => m.status === 'sending' && m.conversationId === msg.conversationId && m.content === msg.content && m.senderId === msg.senderId);
   244→      if (sending && sending.id.startsWith('temp-')) {
   245→        // Remove the temp message, the real one will be added by the store's addMessage
   246→        useChatStore.getState().removeMessage(sending.id);
   247→      }
   248→    };
   249→
   250→    const onReaction = (data: any) => {
   251→      if (data.removed) {
   252→        removeReaction(data.messageId, data.userId, data.emoji);
   253→      } else {
   254→        addReaction(data.messageId, data);
   255→      }
   256→    };
   257→
   258→    socket.on('message:delivered', onDelivered);
   259→    socket.on('message:read', onRead);
   260→    socket.on('message:new', onNew);
   261→    socket.on('reaction:update', onReaction);
   262→
   263→    return () => {
   264→      socket.off('message:delivered', onDelivered);
   265→      socket.off('message:read', onRead);
   266→      socket.off('message:new', onNew);
   267→      socket.off('reaction:update', onReaction);
   268→    };
   269→  }, [token, activeConversationId, user?.id, updateMessage, removeMessage, addReaction, removeReaction]);
   270→
   271→  // Load messages when conversation changes
   272→  useEffect(() => {
   273→    if (!activeConversationId || !token) return;
   274→    const isDemo = token.startsWith('demo-');
   275→    if (isDemo) return;
   276→    let cancelled = false;
   277→    (async () => {
   278→      try {
   279→        const msgs = await api<Message[]>('/conversations/' + activeConversationId + '/messages', { token });
   280→        if (!cancelled) setMessages(msgs);
   281→      } catch (err: any) { if (!cancelled) toast.error(err.message); }
   282→    })();
   283→    return () => { cancelled = true; };
   284→  }, [activeConversationId, token]);
   285→
   286→  // Clean up typing timeout on unmount
   287→  useEffect(() => {
   288→    return () => {
   289→      if (typingTimeout.current) clearTimeout(typingTimeout.current);
   290→    };
   291→  }, []);
   292→
   293→  // Scroll to bottom on new messages (only if already at bottom)
   294→  useEffect(() => {
   295→    if (wasAtBottom) {
   296→      setTimeout(() => {
   297→        scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: 'smooth' });
   298→      }, 50);
   299→    }
   300→  }, [messages, isTyping, wasAtBottom]);
   301→
   302→  const sendMessage = useCallback(() => {
   303→    if (!input.trim() || !activeConversationId) return;
   304→    const socket = getSocket();
   305→    const content = input.trim();
   306→    const replyId = replyTo?.id || null;
   307→
   308→    // Optimistic local message
   309→    const tempId = `temp-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   310→    const optimisticMsg: Message = {
   311→      id: tempId,
   312→      conversationId: activeConversationId,
   313→      senderId: user?.id || '',
   314→      content,
   315→      type: 'text',
   316→      status: 'sending',
   317→      replyToId: replyId,
   318→      edited: false,
   319→      deleted: false,
   320→      createdAt: new Date().toISOString(),
   321→      updatedAt: new Date().toISOString(),
   322→      sender: user ? { id: user.id, email: user.email, displayName: user.displayName, username: user.username, avatar: user.avatar, bio: user.bio, status: user.status, online: user.online, lastSeen: user.lastSeen, theme: user.theme, emailVerified: user.emailVerified, showOnline: user.showOnline, showLastSeen: user.showLastSeen, showReadReceipts: user.showReadReceipts, createdAt: user.createdAt, updatedAt: user.updatedAt } : undefined,
   323→      replyTo: replyTo ? { id: replyTo.id, conversationId: replyTo.conversationId, senderId: replyTo.senderId, content: replyTo.content, type: replyTo.type, status: replyTo.status, replyToId: replyTo.replyToId, edited: replyTo.edited, deleted: replyTo.deleted, createdAt: replyTo.createdAt, updatedAt: replyTo.updatedAt } : null,
   324→    };
   325→    addMessage(optimisticMsg);
   326→
   327→    setInput('');
   328→    setReplyTo(null);
   329→    setShowEmoji(false);
   330→    if (inputRef.current) inputRef.current.style.height = 'auto';
   331→
   332→    // Stop typing
   333→    socket.emit('typing:stop', { conversationId: activeConversationId });
   334→    if (typingTimeout.current) clearTimeout(typingTimeout.current);
   335→
   336→    // Emit to server
   337→    socket.emit('message:send', {
   338→      conversationId: activeConversationId,
   339→      content,
   340→      type: 'text',
   341→      replyToId: replyId,
   342→    });
   343→  }, [input, activeConversationId, replyTo, user, addMessage]);
   344→
   345→  const retryMessage = useCallback((msg: Message) => {
   346→    if (!activeConversationId || msg.status !== 'failed') return;
   347→    const socket = getSocket();
   348→    // Update locally to 'sending'
   349→    updateMessage(msg.id, { status: 'sending' });
   350→    socket.emit('message:send', {
   351→      conversationId: activeConversationId,
   352→      content: msg.content,
   353→      type: 'text',
   354→      replyToId: msg.replyToId,
   355→    });
   356→  }, [activeConversationId, updateMessage]);
   357→
   358→  const handleKeyDown = (e: React.KeyboardEvent) => {
   359→    if (e.key === 'Enter' && !e.shiftKey) {
   360→      e.preventDefault();
   361→      if (editingId) { saveEdit(); } else { sendMessage(); }
   362→    }
   363→  };
   364→
   365→  const handleInputChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
   366→    setInput(e.target.value);
   367→    e.target.style.height = 'auto';
   368→    e.target.style.height = Math.min(e.target.scrollHeight, 120) + 'px';
   369→    if (activeConversationId) {
   370→      const socket = getSocket();
   371→      socket.emit('typing:start', { conversationId: activeConversationId });
   372→      if (typingTimeout.current) clearTimeout(typingTimeout.current);
   373→      typingTimeout.current = setTimeout(() => {
   374→        socket.emit('typing:stop', { conversationId: activeConversationId });
   375→      }, 2000);
   376→    }
   377→  };
   378→
   379→  const startEdit = (msg: Message) => {
   380→    setEditingId(msg.id);
   381→    setEditText(msg.content);
   382→    setReplyTo(null);
   383→    setQuickReactMsgId(null);
   384→    inputRef.current?.focus();
   385→  };
   386→
   387→  const saveEdit = async () => {
   388→    if (!editingId || !editText.trim() || !activeConversationId) return;
   389→    try {
   390→      const updated = await api<Message>('/conversations/' + activeConversationId + '/messages', {
   391→        token, method: 'PUT', body: { messageId: editingId, content: editText.trim() },
   392→      });
   393→      updateMessage(editingId, updated);
   394→      setEditingId(null);
   395→      setEditText('');
   396→    } catch (err: any) { toast.error(err.message); }
   397→  };
   398→
   399→  const deleteMessage = async (msgId: string) => {
   400→    if (token?.startsWith('demo-')) { toast.info('Demo mode'); return; }

   402→      await api('/messages/' + msgId, { token, method: 'DELETE' });
   403→      updateMessage(msgId, { content: 'This message was deleted', deleted: true } as Partial<Message>);
   404→    } catch (err: any) { toast.error(err.message); }
   405→  };
   406→
   407→  const copyMessage = (content: string) => {
   408→    navigator.clipboard.writeText(content).then(
   409→      () => toast.success('Copied to clipboard'),
   410→      () => toast.error('Failed to copy')
   411→    );
   412→  };
   413→
   414→  const blockUser = async () => {
   415→    if (!otherUser) return;
   416→    if (token?.startsWith('demo-')) { toast.info('Demo mode'); return; }
   417→    try {
   418→      await api('/blocks/block', { token, body: { userId: otherUser.id } });
   419→      toast.success(`${otherUser.displayName} has been blocked`);
   420→      setActiveConversationId(null);
   421→    } catch (err: any) { toast.error(err.message); }
   422→  };
   423→
   424→  const forwardMessage = (msg: Message) => {
   425→    setForwardMsg(msg);
   426→    setReplyTo(null);
   427→    setQuickReactMsgId(null);
   428→    inputRef.current?.focus();
   429→  };
   430→
   431→  const sendForward = useCallback(() => {
   432→    if (!forwardMsg?.content || !activeConversationId) return;
   433→    const socket = getSocket();
   434→    socket.emit('message:send', {
   435→      conversationId: activeConversationId,
   436→      content: forwardMsg.content,
   437→      type: 'text',
   438→      replyToId: null,
   439→    });
   440→    setForwardMsg(null);
   441→  }, [forwardMsg, activeConversationId]);
   442→
   443→  const handleReact = useCallback((messageId: string, emoji: string) => {
   444→    const socket = getSocket();
   445→    socket.emit('reaction:add', { messageId, emoji });
   446→    setQuickReactMsgId(null);
   447→  }, []);
   448→
   449→  const filteredMessages = searchMsg
   450→    ? messages.filter((m) => m.content.toLowerCase().includes(searchMsg.toLowerCase()))
   451→    : messages;
   452→
   453→  if (!activeConversationId || !otherUser) {
   454→    return (
   455→      <div className="flex h-full flex-col items-center justify-center bg-background">
   456→        <div className="text-center">
   457→          <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-2xl overflow-hidden">
   458→            <Image src="/logo.png" alt="KIVO" width={64} height={64} quality={100} sizes="64px" className="object-contain p-1.5 opacity-30" />
   459→          </div>
   460→          <h3 className="text-sm font-medium text-muted-foreground">Select a conversation</h3>
   461→          <p className="mt-1 text-xs text-muted-foreground/60">Choose a chat to start messaging</p>
   462→        </div>
   463→      </div>
   464→    );
   465→  }
   466→
   467→  return (
   468→    <div className="flex h-full flex-col bg-background">
   469→      {/* ========== CHAT HEADER ========== */}
   470→      <div className="flex items-center gap-3 border-b border-border/30 px-4 py-3 bg-surface-1">
   471→        <button
   472→          onClick={() => { setActiveConversationId(null); setQuickReactMsgId(null); }}
   473→          className="flex h-8 w-8 items-center justify-center rounded-lg text-muted-foreground hover:text-foreground hover:bg-surface-hover transition-all md:hidden"
   474→        >
   475→          <ArrowLeft className="h-4 w-4" />
   476→        </button>
   477→
   478→        <ProfilePreview user={otherUser}>
   479→          <button className="flex items-center gap-3 flex-1 min-w-0 text-left">
   480→            <div className="relative">
   481→              <Avatar className="h-9 w-9">
   482→                <AvatarImage src={otherUser.avatar || undefined} />
   483→                <AvatarFallback className="text-xs bg-primary/10 text-primary">{getInitials(otherUser.displayName || '?')}</AvatarFallback>
