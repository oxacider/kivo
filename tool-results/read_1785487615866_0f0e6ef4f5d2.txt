    38→function MessageStatus({ status, className = '' }: { status: Message['status']; className?: string }) {
    39→  if (status === 'failed') {
    40→    return <AlertCircle className={`h-3.5 w-3.5 text-destructive ${className}`} />;
    41→  }
    42→  if (status === 'sending') {
    43→    return <div className={`flex items-center gap-0.5 ${className}`}>
    44→      {[0, 1, 2].map(i => (
    45→        <span key={i} className="h-1 w-1 rounded-full bg-current opacity-40"
    46→          style={{ animation: 'kivo-status-pulse 1s ease-in-out infinite', animationDelay: `${i * 0.2}s` }} />
    47→      ))}
    48→    </div>;
    49→  }
    50→  if (status === 'read') {
    51→    return <CheckCheck className={`h-3.5 w-3.5 text-primary ${className}`} />;
    52→  }
    53→  if (status === 'delivered') {
    54→    return <CheckCheck className={`h-3.5 w-3.5 text-muted-foreground/50 ${className}`} />;

    59→// --- Reaction Badge on message ---
    60→function MessageReactions({
    61→  reactions,
    62→  myUserId,
    63→  onReact,
    64→  onLongPress,
    65→}: {
    66→  reactions: Reaction[];
    67→  myUserId: string;
    68→  onReact: (emoji: string) => void;
    69→  onLongPress?: () => void;
    70→}) {
    71→  if (!reactions || reactions.length === 0) return null;
    72→
    73→  const grouped = reactions.reduce<Record<string, Reaction[]>>((acc, r) => {
    74→    if (!acc[r.emoji]) acc[r.emoji] = [];
    75→    acc[r.emoji].push(r);
    76→    return acc;
    77→  }, {});
    78→
    79→  return (
    80→    <div className="flex flex-wrap gap-1 mt-0.5">
    81→      {Object.entries(grouped).map(([emoji, rs]) => {
    82→        const isMine = rs.some(r => r.userId === myUserId);
    83→        const count = rs.length;
    84→        const names = rs.map(r => r.user?.displayName).filter(Boolean).join(', ');
    85→        return (
    86→          <motion.button
    87→            key={emoji}
    88→            initial={{ scale: 0, opacity: 0 }}
    89→            animate={{ scale: 1, opacity: 1 }}
    90→            transition={{ type: 'spring', stiffness: 500, damping: 25 }}
    91→            whileTap={{ scale: 0.85 }}
    92→            onClick={(e) => { e.stopPropagation(); onReact(emoji); }}
    93→            title={names}
    94→            className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-xs transition-colors ${
    95→              isMine
    96→                ? 'bg-primary/15 ring-1 ring-primary/30'
    97→                : 'bg-surface-2 hover:bg-surface-hover'
    98→            }`}
    99→          >
   100→            <span className="text-sm leading-none">{emoji}</span>
   101→            {count > 1 && <span className="text-[10px] font-medium text-muted-foreground">{count}</span>}
   102→          </motion.button>
   103→        );
   104→      })}
   105→    </div>
   106→  );
   107→}

   258→export function ConversationView() {

   271→  const inputRef = useRef<HTMLTextAreaElement>(null);
   272→  const fileInputRef = useRef<HTMLInputElement>(null);
   273→  const typingTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
   274→  const [wasAtBottom, setWasAtBottom] = useState(true);
   275→  const [pendingImage, setPendingImage] = useState<PendingImage | null>(null);
   276→  const [uploadProgress, setUploadProgress] = useState(0);
   277→  const [isUploading, setIsUploading] = useState(false);
   278→
   279→  const activeConv = conversations.find((c) => c.id === activeConversationId);
   280→  const otherUser = activeConv?.otherUser;
   281→  const activeTypingUsers = useMemo(
   282→    () => typingUsers.filter(t => t.conversationId === activeConversationId && t.isTyping && t.userId !== user?.id),
   283→    [typingUsers, activeConversationId, user?.id]
   284→  );
   285→  const isTyping = activeTypingUsers.length > 0;
   286→  const typingDisplay = activeTypingUsers.length === 1
   287→    ? activeTypingUsers[0]
   288→    : null;
   289→
   290→  // Track scroll position
   291→  const handleScroll = useCallback(() => {
   292→    const el = scrollRef.current;
   293→    if (!el) return;
   294→    const atBottom = el.scrollHeight - el.scrollTop - el.clientHeight < 80;
   295→    setWasAtBottom(atBottom);
   296→  }, []);
   297→
   298→  // Socket listeners for delivered, read, reactions, and optimistic message replacement
   299→  useEffect(() => {
   300→    if (!token || token.startsWith('demo-')) return;
   301→    const socket = getSocket();
   302→
   303→    const onDelivered = (data: { messageId: string; conversationId: string; status: string }) => {
   304→      if (data.conversationId !== activeConversationId) return;
   305→      updateMessage(data.messageId, { status: 'delivered' });
   306→    };
   307→
   308→    const onRead = (data: { conversationId: string; userId: string }) => {
   309→      if (data.conversationId !== activeConversationId) return;
   310→      // Batch: mark all sent/delivered/sending messages as read in one update
   311→      const msgs = useChatStore.getState().messages;
   312→      const updates = msgs
   313→        .filter(m => m.senderId === user?.id && (m.status === 'sent' || m.status === 'delivered' || m.status === 'sending'))
   314→        .map(m => ({ id: m.id, status: 'read' as const }));
   315→      if (updates.length > 0) {
   316→        useChatStore.setState((state) => ({
   317→          messages: state.messages.map((m) => {
   318→            const u = updates.find((up) => up.id === m.id);
   319→            return u ? { ...m, status: u.status } : m;
   320→          }),
   321→        }));
   322→      }
   323→    };
   324→
   325→    const onNew = (msg: any) => {
   326→      // Replace optimistic sending message with the real server message
   327→      const msgs = useChatStore.getState().messages;
   328→      // For text messages: match by content
   329→      const textMatch = msgs.find(m => m.status === 'sending' && m.conversationId === msg.conversationId && m.content === msg.content && m.senderId === msg.senderId && !m.attachments?.length);
   330→      // For image messages: match by type + sender + conversation + no real id
   331→      const imageMatch = msgs.find(m => m.status === 'sending' && m.conversationId === msg.conversationId && m.type === 'image' && m.senderId === msg.senderId && m.id.startsWith('temp-'));
   332→      const sending = textMatch || imageMatch;
   333→      if (sending && sending.id.startsWith('temp-')) {
   334→        // Remove the temp message, the real one will be added by the store's addMessage
   335→        useChatStore.getState().removeMessage(sending.id);
   336→      }
   337→    };
   338→
   339→    const onReaction = (data: any) => {
   340→      if (data.removed) {
   341→        removeReaction(data.messageId, data.userId, data.emoji);
   342→      } else {
   343→        addReaction(data.messageId, data);
   344→      }
   345→    };
   346→
   347→    socket.on('message:delivered', onDelivered);
   348→    socket.on('message:read', onRead);
   349→    socket.on('message:new', onNew);
   350→    socket.on('reaction:update', onReaction);
   351→
   352→    return () => {
   353→      socket.off('message:delivered', onDelivered);
   354→      socket.off('message:read', onRead);
   355→      socket.off('message:new', onNew);
   356→      socket.off('reaction:update', onReaction);
   357→    };
   358→  }, [token, activeConversationId, user?.id, updateMessage, removeMessage, addReaction, removeReaction]);
   359→
   360→  // Load messages when conversation changes
   361→  useEffect(() => {

   363→    const isDemo = token.startsWith('demo-');

   367→      try {
   368→        const msgs = await api<Message[]>('/conversations/' + activeConversationId + '/messages', { token });
   369→        if (!cancelled) setMessages(msgs);
   370→      } catch (err: any) { if (!cancelled) toast.error(err.message); }
