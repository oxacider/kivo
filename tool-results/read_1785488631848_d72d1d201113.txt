   396→    // Optimistic local message

   399→      id: tempId,
   400→      conversationId: activeConversationId,
   401→      senderId: user?.id || '',
   402→      content,
   403→      type: 'text',
   404→      status: 'sending',
   405→      replyToId: replyId,

   408→      createdAt: new Date().toISOString(),

   412→    };

   416→    setReplyTo(null);
   417→    setShowEmoji(false);

   419→

   421→    socket.emit('typing:stop', { conversationId: activeConversationId });
   422→    if (typingTimeout.current) clearTimeout(typingTimeout.current);
   423→
   424→    // Emit to server
   425→    socket.emit('message:send', {
   426→      conversationId: activeConversationId,
   427→      content,
   428→      type: 'text',
   429→      replyToId: replyId,
   430→    });
   431→  }, [input, activeConversationId, replyTo, user, addMessage]);

   433→  // --- Image selection handler ---
   434→  const handleImageSelect = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {

   500→      sender: user ? { id: user.id, email: user.email, displayName: user.displayName, username: user.username, avatar: user.avatar, bio: user.bio, status: user.status, online: user.online, lastSeen: user.lastSeen, theme: user.theme, emailVerified: user.emailVerified, showOnline: user.showOnline, showLastSeen: user.showLastSeen, showReadReceipts: user.showReadReceipts, createdAt: user.createdAt, updatedAt: user.updatedAt } : undefined,

   561→        }
   562→      } else {
   563→        setMessageStatus(tempId, 'failed');
