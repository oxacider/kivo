    57→// --- Message Status Icon ---
    58→function MessageStatus({ status, className = '' }: { status: Message['status']; className?: string }) {
    59→  if (status === 'failed') {
    60→    return <AlertCircle className={`h-3.5 w-3.5 text-destructive ${className}`} />;
    61→  }
    62→  if (status === 'queued') {
    63→    return <Clock className={`h-3.5 w-3.5 text-amber-500 ${className}`} />;
    64→  }
    65→  if (status === 'sending') {
    66→    return <div className={`flex items-center gap-0.5 ${className}`}>
    67→      {[0, 1, 2].map(i => (
    68→        <span key={i} className="h-1 w-1 rounded-full bg-current opacity-40"
    69→          style={{ animation: 'kivo-status-pulse 1s ease-in-out infinite', animationDelay: `${i * 0.2}s` }} />
    70→      ))}
    71→    </div>;
    72→  }
    73→  if (status === 'read') {
    74→    return <CheckCheck className={`h-3.5 w-3.5 text-primary ${className}`} />;
    75→  }
    76→  if (status === 'delivered') {
    77→    return <CheckCheck className={`h-3.5 w-3.5 text-muted-foreground/50 ${className}`} />;
    78→  }
    79→  return <Check className={`h-3 w-3 text-muted-foreground/40 ${className}`} />;
    80→}

    82→// --- Reaction Badge on message ---
    83→function MessageReactions({
    84→  reactions,
    85→  myUserId,
    86→  onReact,
    87→  onLongPress,
    88→}: {
    89→  reactions: Reaction[];
    90→  myUserId: string;
    91→  onReact: (emoji: string) => void;
    92→  onLongPress?: () => void;
    93→}) {
    94→  if (!reactions || reactions.length === 0) return null;
    95→
    96→  const grouped = reactions.reduce<Record<string, Reaction[]>>((acc, r) => {
    97→    if (!acc[r.emoji]) acc[r.emoji] = [];
    98→    acc[r.emoji].push(r);
    99→    return acc;
   100→  }, {});
   101→
   102→  return (
   103→    <div className="flex flex-wrap gap-1 mt-0.5">
   104→      {Object.entries(grouped).map(([emoji, rs]) => {
   105→        const isMine = rs.some(r => r.userId === myUserId);
   106→        const count = rs.length;
   107→        const names = rs.map(r => r.user?.displayName).filter(Boolean).join(', ');
   108→        return (
   109→          <motion.button
   110→            key={emoji}
   111→            initial={{ scale: 0, opacity: 0 }}
   112→            animate={{ scale: 1, opacity: 1 }}
   113→            transition={{ type: 'spring', stiffness: 500, damping: 25 }}
   114→            whileTap={{ scale: 0.85 }}
   115→            onClick={(e) => { e.stopPropagation(); onReact(emoji); }}
   116→            title={names}
   117→            className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-xs transition-colors ${
   118→              isMine
   119→                ? 'bg-primary/15 ring-1 ring-primary/30'
   120→                : 'bg-surface-2 hover:bg-surface-hover'
   121→            }`}
   122→          >
   123→            <span className="text-sm leading-none">{emoji}</span>
   124→            {count > 1 && <span className="text-[10px] font-medium text-muted-foreground">{count}</span>}
   125→          </motion.button>
   126→        );
   127→      })}
   128→    </div>
   129→  );
   130→}

   281→export function ConversationView() {
   282→  const { activeConversationId, conversations, messages, setMessages, prependMessages, addMessage, updateMessage, removeMessage, typingUsers, setActiveConversationId, updateConversation, addReaction, removeReaction, setMessageStatus, hasMoreMessages, isLoadingMoreMessages, setLoadingMoreMessages, setHasMoreMessages, networkStatus, isSyncing } = useChatStore();
   283→  const { user, token } = useAuthStore();
   284→  const [input, setInput] = useState('');
   285→  const [showEmoji, setShowEmoji] = useState(false);
   286→  const [replyTo, setReplyTo] = useState<Message | null>(null);
   287→  const [editingId, setEditingId] = useState<string | null>(null);
   288→  const [editText, setEditText] = useState('');
   289→  const [forwardMsg, setForwardMsg] = useState<Message | null>(null);
   290→  const [showSearch, setShowSearch] = useState(false);
   291→  const [searchMsg, setSearchMsg] = useState('');
   292→  const [quickReactMsgId, setQuickReactMsgId] = useState<string | null>(null);
   293→  const scrollRef = useRef<HTMLDivElement>(null);
   294→  const inputRef = useRef<HTMLTextAreaElement>(null);
   295→  const fileInputRef = useRef<HTMLInputElement>(null);
   296→  const typingTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
   297→  const [wasAtBottom, setWasAtBottom] = useState(true);
   298→  const [pendingImage, setPendingImage] = useState<PendingImage | null>(null);
   299→  const [uploadProgress, setUploadProgress] = useState(0);
   300→  const [isUploading, setIsUploading] = useState(false);
   301→
   302→  const activeConv = conversations.find((c) => c.id === activeConversationId);
   303→  const otherUser = activeConv?.otherUser;
   304→  const activeTypingUsers = useMemo(
   305→    () => typingUsers.filter(t => t.conversationId === activeConversationId && t.isTyping && t.userId !== user?.id),
   306→    [typingUsers, activeConversationId, user?.id]
   307→  );
   308→  const isTyping = activeTypingUsers.length > 0;
   309→  const typingDisplay = activeTypingUsers.length === 1
   310→    ? activeTypingUsers[0]
   311→    : null;
   312→
   313→  // Track scroll position + trigger pagination
   314→  const loadMoreTriggeredRef = useRef(false);
   315→  const handleScroll = useCallback(() => {
   316→    const el = scrollRef.current;
   317→    if (!el) return;
   318→    const atBottom = el.scrollHeight - el.scrollTop - el.clientHeight < 80;
   319→    setWasAtBottom(atBottom);
   320→    // Trigger load-more when near top
   321→    if (el.scrollTop < 120 && hasMoreMessages && !isLoadingMoreMessages && !loadMoreTriggeredRef.current) {
   322→      loadMoreTriggeredRef.current = true;
   323→    }
   324→  }, [hasMoreMessages, isLoadingMoreMessages]);
   325→
   326→  // Load older messages when near top
   327→  const [loadMoreTick, setLoadMoreTick] = useState(0);
   328→  useEffect(() => {
   329→    if (!loadMoreTriggeredRef.current) return;
   330→    setLoadMoreTick((t) => t + 1);
   331→  }, [hasMoreMessages, isLoadingMoreMessages]);
   332→
   333→  useEffect(() => {
   334→    if (!activeConversationId || !token || !hasMoreMessages || isLoadingMoreMessages) return;
   335→    const convId = activeConversationId;
   336→    const tok = token;
   337→    const firstMsg = useChatStore.getState().messages[0];
   338→    if (!firstMsg || firstMsg.id.startsWith('temp-')) return;
   339→    const prevHeight = scrollRef.current?.scrollHeight ?? 0;
   340→    let cancelled = false;
   341→    setLoadingMoreMessages(true);
   342→    loadMoreTriggeredRef.current = false;
   343→    (async () => {
   344→      try {
   345→        const result = await api<{ messages: Message[]; hasMore: boolean }>(
   346→          '/conversations/' + convId + '/messages?before=' + encodeURIComponent(firstMsg.createdAt),
   347→          { token: tok }
   348→        );
   349→        if (!cancelled) {
   350→          prependMessages(result.messages);
   351→          setHasMoreMessages(result.hasMore);
   352→          requestAnimationFrame(() => {
   353→            const el = scrollRef.current;
   354→            if (el) el.scrollTop = el.scrollHeight - prevHeight;
   355→          });
   356→        }
   357→      } catch (err: any) {
   358→        if (!cancelled) toast.error(err.message);
   359→      } finally {
   360→        if (!cancelled) setLoadingMoreMessages(false);
   361→      }
   362→    })();
   363→    return () => { cancelled = true; };
   364→  }, [loadMoreTick, activeConversationId, token]);
   365→
   366→  // Socket listeners for delivered, read, reactions, and optimistic message replacement
   367→  useEffect(() => {
   368→    if (!token || token.startsWith('demo-')) return;
   369→    const socket = getSocket();
   370→
   371→    const onDelivered = (data: { messageId: string; conversationId: string; status: string }) => {
   372→      if (data.conversationId !== activeConversationId) return;
   373→      updateMessage(data.messageId, { status: 'delivered' });
   374→    };
   375→
   376→    const onRead = (data: { conversationId: string; userId: string }) => {
   377→      if (data.conversationId !== activeConversationId) return;
   378→      // Batch: mark all sent/delivered/sending messages as read in one update
   379→      const msgs = useChatStore.getState().messages;
   380→      const updates = msgs
   381→        .filter(m => m.senderId === user?.id && (m.status === 'sent' || m.status === 'delivered' || m.status === 'sending'))
   382→        .map(m => ({ id: m.id, status: 'read' as const }));
   383→      if (updates.length > 0) {
   384→        useChatStore.setState((state) => ({
   385→          messages: state.messages.map((m) => {
   386→            const u = updates.find((up) => up.id === m.id);
   387→            return u ? { ...m, status: u.status } : m;
   388→          }),
   389→        }));
   390→      }
   391→    };
   392→
   393→    const onNew = (msg: any) => {
   394→      // Replace optimistic sending message with the real server message
   395→      const msgs = useChatStore.getState().messages;
   396→      // For text messages: match by content
   397→      const textMatch = msgs.find(m => m.status === 'sending' && m.conversationId === msg.conversationId && m.content === msg.content && m.senderId === msg.senderId && !m.attachments?.length);
   398→      // For image messages: match by type + sender + conversation + no real id
   399→      const imageMatch = msgs.find(m => m.status === 'sending' && m.conversationId === msg.conversationId && m.type === 'image' && m.senderId === msg.senderId && m.id.startsWith('temp-'));
   400→      const sending = textMatch || imageMatch;
   401→      if (sending && sending.id.startsWith('temp-')) {
   402→        // Remove the temp message, the real one will be added by the store's addMessage
   403→        useChatStore.getState().removeMessage(sending.id);
   404→      }
   405→    };
   406→
   407→    const onReaction = (data: any) => {
   408→      if (data.removed) {
   409→        removeReaction(data.messageId, data.userId, data.emoji);
   410→      } else {
   411→        addReaction(data.messageId, data);
   412→      }
   413→    };
   414→
   415→    socket.on('message:delivered', onDelivered);
   416→    socket.on('message:read', onRead);
   417→    socket.on('message:new', onNew);
   418→    socket.on('reaction:update', onReaction);
   419→
   420→    return () => {
   421→      socket.off('message:delivered', onDelivered);
   422→      socket.off('message:read', onRead);
   423→      socket.off('message:new', onNew);
   424→      socket.off('reaction:update', onReaction);
   425→    };
   426→  }, [token, activeConversationId, user?.id, updateMessage, removeMessage, addReaction, removeReaction]);
   427→
   428→  // Load latest 30 messages when conversation changes
   429→  useEffect(() => {
   430→    if (!activeConversationId || !token) return;
   431→    const isDemo = token.startsWith('demo-');
   432→    if (isDemo) return;
   433→    let cancelled = false;
   434→    (async () => {
   435→      try {
   436→        const result = await api<{ messages: Message[]; hasMore: boolean }>(
   437→          '/conversations/' + activeConversationId + '/messages',
   438→          { token }
   439→        );
   440→        if (!cancelled) {
   441→          setMessages(result.messages);
   442→          setHasMoreMessages(result.hasMore);
   443→        }
   444→      } catch (err: any) { if (!cancelled) toast.error(err.message); }
   445→    })();
   446→    return () => { cancelled = true; };
   447→  }, [activeConversationId, token]);
   448→
   449→  // Clean up typing timeout on unmount
   450→  useEffect(() => {
   451→    return () => {
   452→      if (typingTimeout.current) clearTimeout(typingTimeout.current);
   453→    };
   454→  }, []);
   455→
   456→  // Scroll to bottom on new messages (only if already at bottom)
   457→  useEffect(() => {
   458→    if (wasAtBottom) {
   459→      setTimeout(() => {
   460→        scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: 'smooth' });
   461→      }, 50);
   462→    }
   463→  }, [messages, isTyping, wasAtBottom]);
   464→
   465→  const sendMessage = useCallback(() => {
   466→    if (!input.trim() || !activeConversationId) return;
   467→    const content = input.trim();
   468→    const replyId = replyTo?.id || null;
   469→
   470→    const tempId = `temp-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   471→    const now = new Date().toISOString();
   472→    const isOnline = isSocketConnected();
   473→    const status: Message['status'] = isOnline ? 'sending' : 'queued';
   474→
   475→    const optimisticMsg: Message = {
   476→      id: tempId,
   477→      conversationId: activeConversationId,
   478→      senderId: user?.id || '',
   479→      content,
   480→      type: 'text',
   481→      status,
   482→      replyToId: replyId,
   483→      edited: false,
   484→      deleted: false,
