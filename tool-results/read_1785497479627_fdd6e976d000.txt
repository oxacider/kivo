    49→function DateDivider({ date }: { date: string }) {
    50→  const d = new Date(date);
    51→  const today = new Date();
    52→  const yesterday = new Date(today);
    53→  yesterday.setDate(yesterday.getDate() - 1);
    54→
    55→  let label: string;
    56→  if (d.toDateString() === today.toDateString()) {
    57→    label = 'Today';
    58→  } else if (d.toDateString() === yesterday.toDateString()) {
    59→    label = 'Yesterday';
    60→  } else {
    61→    label = format(d, 'MMMM d, yyyy');
    62→  }
    63→
    64→  return (
    65→    <div className="flex items-center justify-center my-3">
    66→      <span className="rounded-full bg-surface-2 px-3 py-1 text-[10px] text-muted-foreground/70 font-medium">
    67→        {label}
    68→      </span>
    69→    </div>
    70→  );
    71→}

    82→function MessageStatus({ status, className = '' }: { status: Message['status']; className?: string }) {
    83→  if (status === 'failed') {
    84→    return <AlertCircle className={`h-3.5 w-3.5 text-destructive ${className}`} />;
    85→  }
    86→  if (status === 'queued') {

    88→  }
    89→  if (status === 'sending') {
    90→    return <div className={`flex items-center gap-0.5 ${className}`}>
    91→      {[0, 1, 2].map(i => (
    92→        <span key={i} className="h-1 w-1 rounded-full bg-current opacity-40"
    93→          style={{ animation: 'kivo-status-pulse 1s ease-in-out infinite', animationDelay: `${i * 0.2}s` }} />
    94→      ))}
    95→    </div>;
    96→  }
    97→  if (status === 'read') {
    98→    return <CheckCheck className={`h-3.5 w-3.5 text-primary ${className}`} />;
    99→  }
   100→  if (status === 'delivered') {
   101→    return <CheckCheck className={`h-3.5 w-3.5 text-muted-foreground/50 ${className}`} />;
   102→  }
   103→  return <Check className={`h-3 w-3 text-muted-foreground/40 ${className}`} />;

   106→// --- Reaction Badge on message ---
   107→function MessageReactions({
   108→  reactions,
   109→  myUserId,
   110→  onReact,
   111→  onLongPress,
   112→}: {
   113→  reactions: Reaction[];
   114→  myUserId: string;
   115→  onReact: (emoji: string) => void;
   116→  onLongPress?: () => void;
   117→}) {
   118→  if (!reactions || reactions.length === 0) return null;
   119→
   120→  const grouped = reactions.reduce<Record<string, Reaction[]>>((acc, r) => {
   121→    if (!acc[r.emoji]) acc[r.emoji] = [];
   122→    acc[r.emoji].push(r);
   123→    return acc;
   124→  }, {});
   125→
   126→  return (
   127→    <div className="flex flex-wrap gap-1 mt-0.5">
   128→      {Object.entries(grouped).map(([emoji, rs]) => {
   129→        const isMine = rs.some(r => r.userId === myUserId);
   130→        const count = rs.length;
   131→        const names = rs.map(r => r.user?.displayName).filter(Boolean).join(', ');
   132→        return (
   133→          <motion.button
   134→            key={emoji}
   135→            initial={{ scale: 0, opacity: 0 }}
   136→            animate={{ scale: 1, opacity: 1 }}
   137→            transition={{ type: 'spring', stiffness: 500, damping: 25 }}
   138→            whileTap={{ scale: 0.85 }}
   139→            onClick={(e) => { e.stopPropagation(); onReact(emoji); }}
   140→            title={names}
   141→            className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-xs transition-colors ${
   142→              isMine
   143→                ? 'bg-primary/15 ring-1 ring-primary/30'
   144→                : 'bg-surface-2 hover:bg-surface-hover'
   145→            }`}
   146→          >
   147→            <span className="text-sm leading-none">{emoji}</span>
   148→            {count > 1 && <span className="text-[10px] font-medium text-muted-foreground">{count}</span>}
   149→          </motion.button>
   150→        );
   151→      })}
   152→    </div>
   153→  );
   154→}

   305→export function ConversationView() {
   306→  const { activeConversationId, conversations, messages, setMessages, prependMessages, addMessage, updateMessage, removeMessage, typingUsers, setActiveConversationId, updateConversation, addReaction, removeReaction, setMessageStatus, hasMoreMessages, isLoadingMoreMessages, setLoadingMoreMessages, setHasMoreMessages, networkStatus, isSyncing } = useChatStore();
   307→  const { user, token } = useAuthStore();
   308→  const [input, setInput] = useState('');
   309→  const [showEmoji, setShowEmoji] = useState(false);
   310→  const [replyTo, setReplyTo] = useState<Message | null>(null);
   311→  const [editingId, setEditingId] = useState<string | null>(null);
   312→  const [editText, setEditText] = useState('');
   313→  const [forwardMsg, setForwardMsg] = useState<Message | null>(null);
   314→  const [showSearch, setShowSearch] = useState(false);
   315→  const [searchMsg, setSearchMsg] = useState('');
   316→  const [quickReactMsgId, setQuickReactMsgId] = useState<string | null>(null);
   317→  const scrollRef = useRef<HTMLDivElement>(null);
   318→  const inputRef = useRef<HTMLTextAreaElement>(null);
   319→  const fileInputRef = useRef<HTMLInputElement>(null);
   320→  const typingTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
   321→  const [wasAtBottom, setWasAtBottom] = useState(true);
   322→  const [pendingImage, setPendingImage] = useState<PendingImage | null>(null);
   323→  const [uploadProgress, setUploadProgress] = useState(0);
   324→  const [isUploading, setIsUploading] = useState(false);
   325→
   326→  const activeConv = conversations.find((c) => c.id === activeConversationId);
   327→  const otherUser = activeConv?.otherUser;
   328→  const activeTypingUsers = useMemo(
   329→    () => typingUsers.filter(t => t.conversationId === activeConversationId && t.isTyping && t.userId !== user?.id),
   330→    [typingUsers, activeConversationId, user?.id]
   331→  );
   332→  const isTyping = activeTypingUsers.length > 0;
   333→  const typingDisplay = activeTypingUsers.length === 1
   334→    ? activeTypingUsers[0]
   335→    : null;
   336→
   337→  // Track scroll position + trigger pagination
   338→  const loadMoreTriggeredRef = useRef(false);
   339→  const handleScroll = useCallback(() => {
   340→    const el = scrollRef.current;
   341→    if (!el) return;
   342→    const atBottom = el.scrollHeight - el.scrollTop - el.clientHeight < 80;
   343→    setWasAtBottom(atBottom);
   344→    // Trigger load-more when near top
   345→    if (el.scrollTop < 120 && hasMoreMessages && !isLoadingMoreMessages && !loadMoreTriggeredRef.current) {
   346→      loadMoreTriggeredRef.current = true;
   347→    }
   348→  }, [hasMoreMessages, isLoadingMoreMessages]);
   349→
   350→  // Load older messages when near top
   351→  const [loadMoreTick, setLoadMoreTick] = useState(0);
   352→  useEffect(() => {
   353→    if (!loadMoreTriggeredRef.current) return;
   354→    setLoadMoreTick((t) => t + 1);
   355→  }, [hasMoreMessages, isLoadingMoreMessages]);
   356→
   357→  useEffect(() => {
   358→    if (!activeConversationId || !token || !hasMoreMessages || isLoadingMoreMessages) return;
   359→    const convId = activeConversationId;
   360→    const tok = token;
   361→    const firstMsg = useChatStore.getState().messages[0];
   362→    if (!firstMsg || firstMsg.id.startsWith('temp-')) return;
   363→    const prevHeight = scrollRef.current?.scrollHeight ?? 0;
   364→    let cancelled = false;
   365→    setLoadingMoreMessages(true);
   366→    loadMoreTriggeredRef.current = false;
   367→    (async () => {
   368→      try {
   369→        const result = await api<{ messages: Message[]; hasMore: boolean }>(
   370→          '/conversations/' + convId + '/messages?before=' + encodeURIComponent(firstMsg.createdAt),
   371→          { token: tok }
   372→        );
   373→        if (!cancelled) {
   374→          prependMessages(result.messages);
   375→          setHasMoreMessages(result.hasMore);
   376→          requestAnimationFrame(() => {
   377→            const el = scrollRef.current;
   378→            if (el) el.scrollTop = el.scrollHeight - prevHeight;
   379→          });
   380→        }
   381→      } catch (err: any) {
   382→        if (!cancelled) toast.error(err.message);
   383→      } finally {
   384→        if (!cancelled) setLoadingMoreMessages(false);
   385→      }
   386→    })();
   387→    return () => { cancelled = true; };
   388→  }, [loadMoreTick, activeConversationId, token]);
   389→
   390→  // Socket listeners for delivered, read, reactions, and optimistic message replacement
   391→  useEffect(() => {
   392→    if (!token || token.startsWith('demo-')) return;
   393→    const socket = getSocket();
   394→
   395→    const onDelivered = (data: { messageId: string; conversationId: string; status: string }) => {
   396→      if (data.conversationId !== activeConversationId) return;
   397→      updateMessage(data.messageId, { status: 'delivered' });
   398→    };
   399→
   400→    const onRead = (data: { conversationId: string; userId: string }) => {
   401→      if (data.conversationId !== activeConversationId) return;
   402→      // Batch: mark all sent/delivered/sending messages as read in one update
   403→      const msgs = useChatStore.getState().messages;
   404→      const updates = msgs
   405→        .filter(m => m.senderId === user?.id && (m.status === 'sent' || m.status === 'delivered' || m.status === 'sending' || m.status === 'queued'))
   406→        .map(m => ({ id: m.id, status: 'read' as const }));
   407→      if (updates.length > 0) {
   408→        useChatStore.setState((state) => ({
   409→          messages: state.messages.map((m) => {
   410→            const u = updates.find((up) => up.id === m.id);
   411→            return u ? { ...m, status: u.status } : m;
   412→          }),
   413→        }));
   414→      }
   415→    };
   416→
   417→    const onNew = (msg: any) => {
   418→      // Replace optimistic sending message with the real server message
   419→      const msgs = useChatStore.getState().messages;
   420→      // For text messages: match by content
   421→      const textMatch = msgs.find(m => (m.status === 'sending' || m.status === 'queued') && m.conversationId === msg.conversationId && m.content === msg.content && m.senderId === msg.senderId && !m.attachments?.length);
   422→      // For image messages: match by type + sender + conversation + no real id
   423→      const imageMatch = msgs.find(m => (m.status === 'sending' || m.status === 'queued') && m.conversationId === msg.conversationId && m.type === 'image' && m.senderId === msg.senderId && m.id.startsWith('temp-'));
   424→      const sending = textMatch || imageMatch;
   425→      if (sending && sending.id.startsWith('temp-')) {
   426→        // Remove the temp message, the real one will be added by the store's addMessage
   427→        useChatStore.getState().removeMessage(sending.id);
   428→      }
   429→    };
   430→
   431→    const onReaction = (data: any) => {
   432→      if (data.removed) {
   433→        removeReaction(data.messageId, data.userId, data.emoji);
   434→      } else {
   435→        addReaction(data.messageId, data);
   436→      }
   437→    };
   438→
   439→    socket.on('message:delivered', onDelivered);
   440→    socket.on('message:read', onRead);
   441→    socket.on('message:new', onNew);
   442→    socket.on('reaction:update', onReaction);
   443→
   444→    return () => {
   445→      socket.off('message:delivered', onDelivered);
   446→      socket.off('message:read', onRead);
   447→      socket.off('message:new', onNew);
   448→      socket.off('reaction:update', onReaction);
   449→    };
   450→  }, [token, activeConversationId, user?.id, updateMessage, removeMessage, addReaction, removeReaction]);
   451→
   452→  // Track whether this is the initial load for current conversation (for instant scroll)
   453→  const initialLoadDoneRef = useRef(false);
   454→
   455→  // Load latest 30 messages when conversation changes
   456→  useEffect(() => {
   457→    if (!activeConversationId || !token) return;
   458→    const isDemo = token.startsWith('demo-');
   459→    if (isDemo) return;
   460→    initialLoadDoneRef.current = false;
   461→    let cancelled = false;
   462→    (async () => {
   463→      try {
   464→        const result = await api<{ messages: Message[]; hasMore: boolean }>(
   465→          '/conversations/' + activeConversationId + '/messages',
   466→          { token }
   467→        );
   468→        if (!cancelled) {
   469→          setMessages(result.messages);
   470→          setHasMoreMessages(result.hasMore);
   471→          // Instant scroll to bottom on initial load (not smooth)
   472→          requestAnimationFrame(() => {
   473→            const el = scrollRef.current;
   474→            if (el) {
   475→              el.scrollTop = el.scrollHeight;
   476→              initialLoadDoneRef.current = true;
   477→            }
   478→          });
   479→        }
   480→      } catch (err: any) { if (!cancelled) toast.error(err.message); }
   481→    })();
   482→    return () => { cancelled = true; };
   483→  }, [activeConversationId, token]);
